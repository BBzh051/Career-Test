        // 初始化测试
        function startTest() {
            document.getElementById('intro-page').classList.remove('active');
            document.getElementById('test-page').classList.add('active');
            showQuestion();
        }

        // 显示题目
        function showQuestion() {
            const q = questions[currentQ];
            document.getElementById('current-q').textContent = currentQ + 1;
            document.getElementById('question-text').textContent = q.text;
            
            // 创建选项按钮 (1~5)
            const optionsContainer = document.getElementById('options-container');
            optionsContainer.innerHTML = '';
            
            for (let i = 1; i <= 5; i++) {
                const option = document.createElement('div');
                option.className = 'option';
                option.textContent = i;
                option.dataset.value = i;
                
                // 如果已答过，显示选中状态
                if (answers[q.id] === i) {
                    option.classList.add('selected');
                }
                
                option.onclick = function() {
                    // 清除其他选项的选中
                    document.querySelectorAll('.option').forEach(opt => {
                        opt.classList.remove('selected');
                    });
                    this.classList.add('selected');
                    answers[q.id] = i;
                    document.getElementById('next-btn').disabled = false;
                };
                optionsContainer.appendChild(option);
            }
            
            // 更新进度条
            const progress = ((currentQ + 1) / questions.length) * 100;
            document.getElementById('progress').style.width = progress + '%';
        }

        // 下一题
        function nextQuestion() {
            if (currentQ < questions.length - 1) {
                currentQ++;
                showQuestion();
                // 重置下一题按钮状态
                document.getElementById('next-btn').disabled = true;
            } else {
                // 测试完成，显示结果
                calculateResult();
            }
        }

        // 计算结果
        function calculateResult() {
            // 初始化维度分数
            const dimensions = {
                "逻辑分析力": 0,
                "创造力与艺术天赋": 0,
                "人际影响力": 0,
                "动手实践力": 0,
                "语言与表达力": 0,
                "领导力与决策力": 0
            };
            const dimCount = {
                "逻辑分析力": 0,
                "创造力与艺术天赋": 0,
                "人际影响力": 0,
                "动手实践力": 0,
                "语言与表达力": 0,
                "领导力与决策力": 0
            };

            // 统计每道题的答案
            for (const qid in answers) {
                const dim = dimensionMap[qid];
                dimensions[dim] += answers[qid];
                dimCount[dim]++;
            }

            // 计算平均分（转换为百分制，满分100）
            const result = {};
            for (const dim in dimensions) {
                const avg = dimensions[dim] / dimCount[dim]; // 平均分 (1~5)
                result[dim] = Math.round((avg / 5) * 100); // 转为百分比
            }

            // 排序，找出TOP3
            const sorted = Object.entries(result).sort((a, b) => b[1] - a[1]);
            const top3 = sorted.slice(0, 3).map(item => ({name: item[0], score: item[1]}));

            // 显示结果
            showResult(result, top3);
        }

        // 显示结果页

            document.getElementById('test-page').classList.remove('active');
            document.getElementById('result-page').classList.add('active');

            // 维度分数条
            const dimResultHTML = '';
            for (const [dim, score] of Object.entries(result)) {
                dimResultHTML += `
                <div class="result-card">
                    <div class="dimension">${dim}</div>
                    <div class="score-bar">
                        <div class="score-fill" style="width: ${score}%">${score}分</div>
                    </div>
                </div>
                `;
            }
            document.getElementById('dimensions-result').innerHTML = dimResultHTML;

            // TOP3
            let top3HTML = '';
            top3.forEach((item, index) => {
                let medal = '';
                if(index===0) medal = '🥇';
                else if(index===1) medal = '🥈';
                else if(index===2) medal = '🥉';
                top3HTML += `<p><strong>${medal} ${item.name}</strong> — ${item.score}分</p>`;
            });
            document.getElementById('top3-list').innerHTML = top3HTML;

            // 个性化建议
            const adviceMap = {
                "逻辑分析力": "你的分析能力突出！建议深入数据分析、工程、科研、金融建模等领域。可学习Python、统计学、逻辑学，考取CFA、精算师等证书。",
                "创造力与艺术天赋": "天生的创意大师！适合设计、广告、影视、艺术创作、创业创新。多参与跨界项目，建立作品集，关注行业设计奖项。",
                "人际影响力": "你善于理解他人！教育、心理咨询、社工、人力资源、销售管理是你的强项。提升倾听技巧，学习NLP或心理学课程。",
                "动手实践力": "动手能力一流！机械、电子、建筑、工匠类工作、实验室技术员都很适合。考取工信部认证证书，积累实操项目经验。",
                "语言与表达力": "表达大师！媒体、翻译、法律、教师、公关文案是你的舞台。精进外语能力，学习播音主持或法律文书写作。",
                "领导力与决策力": "天生的领导者！企业管理、创业、项目管理、政府事务是你的方向。学习MBA课程，锻炼战略思维，积累团队管理经验。"
            };

            let adviceText = `基于你的<strong>${top3[0].name}</strong>优势，导师建议：<br><br>${adviceMap[top3[0].name]}<br><br>`;
            adviceText += `同时，你的<strong>${top3[1].name}</strong>和<strong>${top3[2].name}</strong>也很突出，建议在职业规划中结合这三大优势，形成独特的竞争力组合！`;
            
            document.getElementById('advice-text').innerHTML = adviceText;
        }
